# 基本配置

```json
{
    /* 通用配置 */
    // 颜色主题
    "workbench.colorTheme": "Coldark - Cold",
    // 关闭预览模式. 默认打开, 单击文件只能打开一个
    "workbench.editor.enablePreview": false,
    // 设置字体
    "editor.fontFamily": "'JetBrains Mono NL Medium'",
    "editor.fontSize": 12,
    // 关闭右侧小地图
    "editor.minimap.enabled": false,
    // 图标主题
    "workbench.iconTheme": "vscode-icons",
    // 关闭黏性行
    "editor.stickyScroll.enabled": false,
    "workbench.tree.enableStickyScroll": false,
    "terminal.integrated.stickyScroll.enabled": false,
    // 展开文件夹, 即使只有一个子文件
    "explorer.compactFolders": false,
    // 目录缩进 15 像素
    "workbench.tree.indent": 15,
    // 始终显示目录辅助线
    "workbench.tree.renderIndentGuides": "always",
    "scm.compactFolders": false,
    // 重启后不打开任何文件
    "window.restoreWindows": "none",
    // 关闭文件信任提示
    "security.workspace.trust.untrustedFiles": "open",
    "remote.SSH.remotePlatform": {
        "192.168.56.101": "linux",
        "192.168.56.103": "linux",
        "192.168.56.102": "linux",
        "ubuntu-work": "linux",
        "192.168.56.10": "linux",
        "vagrant-cppstudy": "linux"
    },
    "[xml]": {
        "editor.defaultFormatter": "redhat.vscode-xml"
    },
    "extensions.ignoreRecommendations": true,
    
}
```
